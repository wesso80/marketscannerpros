import { beforeEach, expect, it, vi } from 'vitest';
import { NextRequest } from 'next/server';
const h=vi.hoisted(()=>({auth:vi.fn(),setAuto:vi.fn(),state:vi.fn(),orders:vi.fn(),stamp:vi.fn(),set:vi.fn(),get:vi.fn()}));
vi.mock('@/lib/adminAuth',()=>({requireAdmin:h.auth}));
vi.mock('@/lib/admin/cryptoAutomation',()=>({setCryptoAutomation:h.setAuto,cryptoAutomationState:vi.fn(async()=>({enabled:true}))}));
vi.mock('@/lib/admin/cryptoPaper',()=>({cryptoPaperState:h.state,setCryptoPaperActive:vi.fn(),runCryptoPaperCycle:vi.fn(),cryptoPaperTradeLog:vi.fn()}));
vi.mock('@/lib/admin/cryptoPaperBase',()=>({cryptoBaseSleeveState:vi.fn(async()=>({portfolio:null})),ensureBaseSleeve:vi.fn(),runCryptoBaseSleeveCycle:vi.fn()}));
vi.mock('@/lib/admin/portfolio-lab/portfolioStore',()=>({getDefaultPortfolio:vi.fn(async()=>({id:'p',workspaceId:'w'}))}));
vi.mock('@/lib/admin/portfolio-lab/simulatedOrderEngine',()=>({createSimulatedOrder:h.orders}));
vi.mock('@/lib/admin/portfolio-lab/positionSizing',()=>({sizeForPortfolio:vi.fn(async()=>({ok:true,quantity:1,notional:100,riskDollars:10}))}));
vi.mock('@/lib/admin/portfolio-lab/riskEngine',()=>({checkPreTrade:vi.fn(async()=>({ok:true,warnings:[]}))}));
vi.mock('@/lib/admin/portfolio-lab/journalEngine',()=>({writeJournal:vi.fn()}));
vi.mock('@/lib/redis',()=>({getRedis:()=>({get:h.get,set:h.set})}));
vi.mock('@/lib/admin/cryptoFlow',()=>({stampMomentumVolume:h.stamp,unavailableFlowStamp:vi.fn()}));
vi.mock('@/lib/admin/cryptoMomentumBatch',()=>({runMomentumBatch:vi.fn()}));
import { POST as paper } from '@/app/api/admin/crypto-markets/paper/route';
import { POST as order } from '@/app/api/admin/portfolio-lab/create-sim-order/route';
import { GET as momentum } from '@/app/api/admin/crypto-markets/momentum/route';
const req=(path:string,body:unknown,origin='https://foreign.example')=>new NextRequest('https://marketscannerpros.app'+path,{method:'POST',headers:{origin,'content-type':'application/json','idempotency-key':'same-request'},body:JSON.stringify(body)});
beforeEach(()=>{vi.clearAllMocks();process.env.CRYPTO_MARKETS_PAUSED='false';h.auth.mockResolvedValue({ok:true,workspaceId:'w'});h.state.mockResolvedValue({portfolio:null});h.orders.mockResolvedValue({id:'new-order'});h.get.mockImplementation(async(key:string)=>key.endsWith(':batch-lock')?null:{rows:[{stage:'MOMENTUM_VOLUME',pair:{product:'BTC-USD'}}]});});
it('paper global automation accepts a foreign Origin after successful authentication',async()=>{expect((await paper(req('/api/admin/crypto-markets/paper',{action:'auto_enable'}))).status).toBe(200);expect(h.setAuto).toHaveBeenCalledWith(true);});
it('two workspace admins address the same global automation setter',async()=>{await paper(req('/api/admin/crypto-markets/paper',{action:'auto_enable'}));h.auth.mockResolvedValue({ok:true,workspaceId:'other'});await paper(req('/api/admin/crypto-markets/paper',{action:'auto_pause'}));expect(h.setAuto.mock.calls).toEqual([[true],[false]]);});
it('an accepted global change followed by a failed account read returns 503',async()=>{h.state.mockRejectedValueOnce(Error('snapshot unavailable'));expect((await paper(req('/api/admin/crypto-markets/paper',{action:'auto_enable'}))).status).toBe(503);expect(h.setAuto).toHaveBeenCalledWith(true);});
it('repeated identical manual-order requests dispatch two creates despite an idempotency key',async()=>{const body={symbol:'AAPL',assetClass:'equity',side:'LONG',entry:100,stop:90};for(let i=0;i<2;i++)expect((await order(req('/api/admin/portfolio-lab/create-sim-order',body))).status).toBe(200);expect(h.orders).toHaveBeenCalledTimes(2);});
it('momentum GET can stamp provider-derived data and write the shared scan',async()=>{expect((await momentum(new Request('https://marketscannerpros.app/api/admin/crypto-markets/momentum'))).status).toBe(200);expect(h.stamp).toHaveBeenCalledTimes(1);expect(h.set).toHaveBeenCalledWith('admin:crypto-markets:momentum-volume:v1',expect.any(Object),{ex:86400});});
it('two simultaneous momentum reads both pass the batch-lock presence check',async()=>{await Promise.all([momentum(new Request('https://marketscannerpros.app/api/admin/crypto-markets/momentum')),momentum(new Request('https://marketscannerpros.app/api/admin/crypto-markets/momentum'))]);expect(h.stamp).toHaveBeenCalledTimes(2);expect(h.set).toHaveBeenCalledTimes(2);});
